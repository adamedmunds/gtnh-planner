# GTNH Planner — Plan 2: 3D Editor

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A browser editor (Vite + React + react-three-fiber) where the user places, rotates and connects GT machines, cables and pipes in 3D with real GTNH textures, sees a materials list, saves/loads builds, loads the oil berry line as an example, and exports a Schematica `.schematic`.

**Architecture:** All editing rules are pure TypeScript (`src/model/editing.ts`, `src/model/buildJson.ts`, `src/editor/state.ts` reducer) and unit-tested; React components are thin views over the reducer. The 3D scene renders blocks from the catalog with textures extracted from the user's GregTech jar by a one-time script (`npm run textures`); missing textures fall back to flat colours. A Playwright test drives the real app end to end.

**Tech Stack:** React 19, Vite 8, TypeScript, three 0.186, @react-three/fiber 9, @react-three/drei 10, fflate, Vitest, @playwright/test.

**Spec:** `docs/superpowers/specs/2026-10-18-gtnh-planner-design.md` (sections 1 Texture extractor, 4 Editor, 5 Exporter, milestones 4–6, Error handling)

## Global Constraints

- React for all UI (user preference); no backend.
- Pure modules (`src/catalog`, `src/model`, `src/export`, `src/presets`, `src/editor/state.ts`, `src/editor/persistence.ts`, `src/editor/download.ts`, `src/editor/scene/geometry.ts`) must not import React or three.
- Textures are extracted from the user's own jar into `public/textures/` (git-ignored) and never committed; texture paths are relative to `assets/gregtech/textures/` in the jar (e.g. `blocks/iconsets/MACHINE_LV_SIDE.png`).
- GTNH instance default: `C:\Users\adam\AppData\Roaming\PrismLauncher\instances\GT_New_Horizons_2.9.0-RC-1_Java_17-26\.minecraft` (override with `--instance <path>` or `GTNH_INSTANCE`).
- GT side order everywhere: 0 down, 1 up, 2 north (−Z), 3 south (+Z), 4 west (−X), 5 east (+X).
- Controls (spec §4): left-click = place against the clicked face; right-click = remove; `R` = rotate hovered machine (N→E→S→W), `Shift+R` includes up/down; `Shift`+left-click on a cable/pipe face = toggle that side; drag = orbit, right/middle-drag = pan, scroll = zoom; a click that moved more than 4 px is a drag, not a click.
- Build autosaves to `localStorage` key `gtnh-planner:build` (debounced, try/catch); Save/Load `.json`; Export disabled when empty.
- Commits end with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

## Review Focus

- **Malformed or hand-edited build JSON** (bad JSON, wrong version, string coordinates, facing 7, junk masks) → clear error or per-block warnings; the current build is untouched on a hard error and bad fields never reach the exporter. Tests in Task 3.
- **Storage unavailable or corrupt** (private window, quota, garbage in `localStorage`) → app starts empty with a notice and keeps working. Tests in Task 6.
- **Orbit-drag ending over a block** → must not place or remove anything. `isClick` test in Task 5.
- **Placing into an occupied cell, rotating a pipe, toggling a side on a machine** → no-ops, never corrupt state. Tests in Task 2/4.
- **Extractor not run / texture missing** → flat colours, no crash, one console warning per texture. Test in Task 5.

```ts
import { CATALOG, type Catalog } from '../catalog';
import { computeConnections } from './connections';
import { posKey } from './directions';
import type { Facing, PlacedBlock, Vec3 } from './types';

export type BlockMap = ReadonlyMap<string, PlacedBlock>;

export const toBlockMap = (blocks: readonly PlacedBlock[]): BlockMap => new Map(blocks.map((b) => [posKey(b.pos), b]));

export function placeBlock(m: BlockMap, block: PlacedBlock): BlockMap {
  const k = posKey(block.pos);
  if (m.has(k)) return m;
  const next = new Map(m);
  next.set(k, block);
  return next;
}

export function removeBlock(m: BlockMap, pos: Vec3): BlockMap {
  const k = posKey(pos);
  if (!m.has(k)) return m;
  const next = new Map(m);
  next.delete(k);
  return next;
}

const HORIZONTAL: readonly Facing[] = [2, 5, 3, 4]; // N → E → S → W
const WITH_VERTICAL: readonly Facing[] = [2, 5, 3, 4, 1, 0];

export function nextFacing(f: Facing | undefined, includeVertical: boolean): Facing {
  const cycle = includeVertical ? WITH_VERTICAL : HORIZONTAL;
  const i = cycle.indexOf(f ?? 3);
  return cycle[(i + 1) % cycle.length];
}

function replace(m: BlockMap, block: PlacedBlock): BlockMap {
  const next = new Map(m);
  next.set(posKey(block.pos), block);
  return next;
}

export function rotateBlock(m: BlockMap, pos: Vec3, includeVertical: boolean, catalog: Catalog = CATALOG): BlockMap {
  const block = m.get(posKey(pos));
  if (!block || catalog.get(block.key)?.kind !== 'machine') return m;
  return replace(m, { ...block, facing: nextFacing(block.facing, includeVertical) });
}

export function toggleSide(m: BlockMap, pos: Vec3, side: number, catalog: Catalog = CATALOG): BlockMap {
  const k = posKey(pos);
  const block = m.get(k);
  const kind = block && catalog.get(block.key)?.kind;
  if (!block || !kind || kind === 'machine' || side < 0 || side > 5) return m;
  const bit = 1 << side;
  const connected = ((computeConnections([...m.values()], catalog).get(k) ?? 0) & bit) !== 0;
  const mask = block.manualMask ?? { forcedOn: 0, forcedOff: 0 };
  const next = connected
    ? { forcedOn: mask.forcedOn & ~bit, forcedOff: mask.forcedOff | bit }
    : { forcedOn: mask.forcedOn | bit, forcedOff: mask.forcedOff & ~bit };
  // If the user's toggle just undoes an earlier override, drop the override entirely.
  const auto = ((computeConnections([...m.values()].map((b) => (b === block ? { ...b, manualMask: undefined } : b)), catalog)
    .get(k) ?? 0) & bit) !== 0;
  const cleaned = {
    forcedOn: auto ? next.forcedOn & ~bit : next.forcedOn,
    forcedOff: auto ? next.forcedOff : next.forcedOff & ~bit,
  };
  const { manualMask: _drop, ...rest } = block;
  void _drop;
  return replace(m, cleaned.forcedOn || cleaned.forcedOff ? { ...rest, manualMask: cleaned } : rest);
}

export function placementFacing(normalSide: number, cameraForward: Vec3): Facing {
  if (normalSide >= 2 && normalSide <= 5) return normalSide as Facing;
  const [fx, , fz] = cameraForward;
  if (Math.abs(fx) > Math.abs(fz)) return fx > 0 ? 4 : 5;
  return fz > 0 ? 2 : 3;
}

export interface BomLine { key: string; name: string; group: string; count: number }

const GROUP_ORDER = ['Machines', 'Power', 'Cables', 'Fluid pipes', 'Item pipes'];

export function billOfMaterials(blocks: Iterable<PlacedBlock>, catalog: Catalog = CATALOG): BomLine[] {
  const lines = new Map<string, BomLine>();
  for (const b of blocks) {
    const e = catalog.get(b.key);
    if (!e) continue;
    const line = lines.get(e.key) ?? { key: e.key, name: e.name, group: e.group, count: 0 };
    line.count++;
    lines.set(e.key, line);
  }
  const rank = (g: string) => (GROUP_ORDER.indexOf(g) + 1 || GROUP_ORDER.length + 1);
  return [...lines.values()].sort((a, b) => rank(a.group) - rank(b.group) || a.name.localeCompare(b.name));
}
```

Note on `toggleSide`'s "cleaned" step: an override bit is only kept when it differs from what auto-connection would do, so toggling a side twice returns the block to having no `manualMask` (the test checks this).

```ts
import type { Build, Facing, ManualMask, PlacedBlock } from './types';

export class BuildParseError extends Error {}

export function serializeBuild(build: Build): string {
  return JSON.stringify(build, null, 2);
}

const isInt = (v: unknown): v is number => typeof v === 'number' && Number.isInteger(v);
const isMaskBits = (v: unknown): v is number => isInt(v) && v >= 0 && v <= 63;

export function parseBuild(text: string): { build: Build; warnings: string[] } {
  let raw: unknown;
  try {
    raw = JSON.parse(text);
  } catch {
    throw new BuildParseError('Not a valid JSON file');
  }
  if (typeof raw !== 'object' || raw === null || Array.isArray(raw) || (raw as { version?: unknown }).version !== 1) {
    throw new BuildParseError('Unsupported build file (expected version 1)');
  }
  const obj = raw as { name?: unknown; blocks?: unknown };
  if (!Array.isArray(obj.blocks)) throw new BuildParseError('Build file has no blocks list');

  const warnings: string[] = [];
  const blocks: PlacedBlock[] = [];
  obj.blocks.forEach((b: unknown, i: number) => {
    const rb = (typeof b === 'object' && b !== null ? b : {}) as Record<string, unknown>;
    const pos = rb.pos;
    if (!Array.isArray(pos) || pos.length !== 3 || !pos.every(isInt)) {
      warnings.push(`Skipped block #${i + 1}: invalid position`);
      return;
    }
    if (typeof rb.key !== 'string' || rb.key === '') {
      warnings.push(`Skipped block #${i + 1}: missing block type`);
      return;
    }
    const block: PlacedBlock = { pos: [pos[0], pos[1], pos[2]], key: rb.key };
    if (rb.facing !== undefined) {
      if (isInt(rb.facing) && rb.facing >= 0 && rb.facing <= 5) block.facing = rb.facing as Facing;
      else warnings.push(`Block #${i + 1}: ignored invalid facing`);
    }
    if (rb.manualMask !== undefined) {
      const m = rb.manualMask as Partial<ManualMask> | null;
      if (m && isMaskBits(m.forcedOn) && isMaskBits(m.forcedOff)) block.manualMask = { forcedOn: m.forcedOn, forcedOff: m.forcedOff };
      else warnings.push(`Block #${i + 1}: ignored invalid connection overrides`);
    }
    blocks.push(block);
  });

  return { build: { version: 1, name: typeof obj.name === 'string' ? obj.name : 'untitled', blocks }, warnings };
}
```

```ts
import { CATALOG } from '../catalog';
import { placeBlock, removeBlock, rotateBlock, toBlockMap, toggleSide, type BlockMap } from '../model/editing';
import type { Build, Facing, Vec3 } from '../model/types';

export const DEFAULT_KEY = 'distillery_lv';

export interface EditorState {
  name: string;
  blocks: BlockMap;
  selectedKey: string;
  /** Highest visible Y level; null shows everything. */
  layer: number | null;
}

export type Action =
  | { type: 'place'; pos: Vec3; facing?: Facing }
  | { type: 'remove'; pos: Vec3 }
  | { type: 'rotate'; pos: Vec3; includeVertical: boolean }
  | { type: 'toggleSide'; pos: Vec3; side: number }
  | { type: 'select'; key: string }
  | { type: 'setName'; name: string }
  | { type: 'setLayer'; layer: number | null }
  | { type: 'load'; build: Build }
  | { type: 'new' };

export function initialState(build?: Build): EditorState {
  return {
    name: build?.name ?? 'untitled',
    blocks: toBlockMap(build?.blocks ?? []),
    selectedKey: DEFAULT_KEY,
    layer: null,
  };
}

const withBlocks = (s: EditorState, blocks: BlockMap) => (blocks === s.blocks ? s : { ...s, blocks });

export function reducer(s: EditorState, a: Action): EditorState {
  switch (a.type) {
    case 'place': {
      const entry = CATALOG.get(s.selectedKey);
      if (!entry) return s;
      const block = entry.kind === 'machine'
        ? { pos: a.pos, key: entry.key, facing: a.facing ?? 3 }
        : { pos: a.pos, key: entry.key };
      return withBlocks(s, placeBlock(s.blocks, block));
    }
    case 'remove':
      return withBlocks(s, removeBlock(s.blocks, a.pos));
    case 'rotate':
      return withBlocks(s, rotateBlock(s.blocks, a.pos, a.includeVertical));
    case 'toggleSide':
      return withBlocks(s, toggleSide(s.blocks, a.pos, a.side));
    case 'select':
      return CATALOG.has(a.key) ? { ...s, selectedKey: a.key } : s;
    case 'setName':
      return { ...s, name: a.name.trim() || 'untitled' };
    case 'setLayer':
      return { ...s, layer: a.layer };
    case 'load':
      return { ...initialState(a.build), selectedKey: s.selectedKey };
    case 'new':
      return { ...initialState(), selectedKey: s.selectedKey };
  }
}

export function toBuild(s: EditorState): Build {
  return { version: 1, name: s.name, blocks: [...s.blocks.values()] };
}
```

```ts
import type { CatalogEntry } from '../src/catalog';

export const JAR_TEXTURE_ROOT = 'assets/gregtech/textures/';

export function wantedTexturePaths(entries: readonly CatalogEntry[]): string[] {
  const set = new Set<string>();
  for (const e of entries) for (const p of Object.values(e.textures)) if (p) set.add(p);
  return [...set].sort();
}

export function selectTextures(
  files: Record<string, Uint8Array>,
  wanted: readonly string[],
): { files: Record<string, Uint8Array>; missing: string[] } {
  const out: Record<string, Uint8Array> = {};
  const missing: string[] = [];
  for (const rel of wanted) {
    const data = files[JAR_TEXTURE_ROOT + rel];
    if (!data) {
      missing.push(rel);
      continue;
    }
    out[rel] = data;
    const meta = files[`${JAR_TEXTURE_ROOT}${rel}.mcmeta`];
    if (meta) out[`${rel}.mcmeta`] = meta;
  }
  return { files: out, missing };
}
```

```ts
import type { Vec3 } from '../../model/types';

export interface Box { center: Vec3; size: Vec3 }

const AXIS_DIRS: readonly [number, number][] = [
  [1, -1], [1, 1], [2, -1], [2, 1], [0, -1], [0, 1], // [axis, sign] per GT side
];

/** Centre cube plus one arm per connected side, in unit-cell coordinates. */
export function conduitBoxes(thickness: number, mask: number): Box[] {
  const t = thickness;
  const boxes: Box[] = [{ center: [0.5, 0.5, 0.5], size: [t, t, t] }];
  const arm = (1 - t) / 2;
  AXIS_DIRS.forEach(([axis, sign], side) => {
    if (!(mask & (1 << side))) return;
    const center: [number, number, number] = [0.5, 0.5, 0.5];
    const size: [number, number, number] = [t, t, t];
    center[axis] = 0.5 + sign * (t / 2 + arm / 2);
    size[axis] = arm;
    boxes.push({ center, size });
  });
  return boxes;
}

export function sideFromNormal(n: Vec3): number {
  const idx = AXIS_DIRS.findIndex(([axis, sign]) => Math.round(n[axis]) === sign && Math.abs(n[axis]) > 0.9);
  return idx;
}

export const isClick = (delta: number) => delta <= 4;

/** GT side shown on each three.js BoxGeometry material slot (+x, -x, +y, -y, +z, -z). */
export const FACE_ORDER: readonly number[] = [5, 4, 1, 0, 3, 2];
```

```ts
import { useEffect, useState } from 'react';
import * as THREE from 'three';

type Loader = (url: string) => Promise<TexImageSource>;

const loadImage: Loader = (url) =>
  new Promise((resolve, reject) => {
    const img = new Image();
    img.onload = () => resolve(img);
    img.onerror = () => reject(new Error(`Failed to load ${url}`));
    img.src = url;
  });

function pixelTexture(source: TexImageSource | HTMLCanvasElement): THREE.Texture {
  const tex = new THREE.Texture(source as never);
  tex.magFilter = THREE.NearestFilter;
  tex.minFilter = THREE.NearestFilter;
  tex.colorSpace = THREE.SRGBColorSpace;
  tex.needsUpdate = true;
  return tex;
}

export function createTextureCache(load: Loader = loadImage, base = '/textures/') {
  const images = new Map<string, Promise<TexImageSource | null>>();
  const textures = new Map<string, Promise<THREE.Texture | null>>();

  function image(rel: string) {
    if (!images.has(rel)) {
      images.set(rel, load(base + rel).catch((err) => {
        console.warn(`Texture unavailable (run "npm run textures"): ${rel}`, err);
        return null;
      }));
    }
    return images.get(rel)!;
  }

  function get(rel?: string): Promise<THREE.Texture | null> {
    if (!rel) return Promise.resolve(null);
    if (!textures.has(rel)) textures.set(rel, image(rel).then((img) => (img ? pixelTexture(img) : null)));
    return textures.get(rel)!;
  }

  /** Base texture with an overlay drawn on top (machine front faces). Animated strips use their first frame. */
  function composite(baseRel?: string, overlayRel?: string): Promise<THREE.Texture | null> {
    if (!baseRel) return Promise.resolve(null);
    if (!overlayRel) return get(baseRel);
    const key = `${baseRel}+${overlayRel}`;
    if (!textures.has(key)) {
      textures.set(key, Promise.all([image(baseRel), image(overlayRel)]).then(([b, o]) => {
        if (!b) return null;
        const size = (b as HTMLImageElement).width || 16;
        const canvas = document.createElement('canvas');
        canvas.width = canvas.height = size;
        const ctx = canvas.getContext('2d')!;
        ctx.imageSmoothingEnabled = false;
        ctx.drawImage(b as CanvasImageSource, 0, 0, size, size, 0, 0, size, size);
        if (o) ctx.drawImage(o as CanvasImageSource, 0, 0, size, size, 0, 0, size, size);
        return pixelTexture(canvas);
      }));
    }
    return textures.get(key)!;
  }

  return { get, composite };
}

export const textureCache = createTextureCache();

export function useAsync<T>(fn: () => Promise<T>, deps: unknown[]): T | undefined {
  const [value, setValue] = useState<T>();
  useEffect(() => {
    let live = true;
    fn().then((v) => live && setValue(v));
    return () => {
      live = false;
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, deps);
  return value;
}
```

```ts
import { parseBuild, serializeBuild } from '../model/buildJson';
import type { Build } from '../model/types';

export const STORAGE_KEY = 'gtnh-planner:build';

export function safeStorage(): Storage | undefined {
  try {
    return typeof window !== 'undefined' ? window.localStorage : undefined;
  } catch {
    return undefined;
  }
}

export function saveToStorage(storage: Storage | undefined, build: Build): boolean {
  if (!storage) return false;
  try {
    storage.setItem(STORAGE_KEY, serializeBuild(build));
    return true;
  } catch {
    return false;
  }
}

export function loadFromStorage(storage: Storage | undefined): { build: Build | null; error?: string } {
  if (!storage) return { build: null };
  let text: string | null;
  try {
    text = storage.getItem(STORAGE_KEY);
  } catch {
    return { build: null };
  }
  if (text === null) return { build: null };
  try {
    return { build: parseBuild(text).build };
  } catch {
    return { build: null, error: "Couldn't restore your last build; starting empty." };
  }
}
```

```tsx
import type { CatalogEntry } from '../../catalog';
import type { PlacedBlock } from '../../model/types';
import { textureCache, useAsync } from '../textures';
import { FACE_ORDER } from './geometry';

export function MachineMesh({ block, entry }: { block: PlacedBlock; entry: CatalogEntry }) {
  const t = entry.textures;
  const tex = useAsync(
    () => Promise.all([textureCache.get(t.top), textureCache.get(t.side), textureCache.get(t.bottom), textureCache.composite(t.side, t.front)]),
    [entry.key],
  );
  const [top, side, bottom, front] = tex ?? [null, null, null, null];
  const facing = block.facing ?? 3;
  const faceTexture = (s: number) => (s === facing ? front ?? side : s === 1 ? top : s === 0 ? bottom : side);
  const [x, y, z] = block.pos;
  return (
    <mesh position={[x + 0.5, y + 0.5, z + 0.5]} userData={{ pos: block.pos }}>
      <boxGeometry args={[1, 1, 1]} />
      {FACE_ORDER.map((s, i) => {
        const map = faceTexture(s);
        const untexturedFront = !map && s === facing;
        return (
          <meshStandardMaterial
            key={i}
            attach={`material-${i}`}
            map={map ?? null}
            color={map ? '#ffffff' : untexturedFront ? '#f0f0f0' : entry.color}
          />
        );
      })}
    </mesh>
  );
}
```

```tsx
import { useMemo } from 'react';
import { Canvas, useThree, type ThreeEvent } from '@react-three/fiber';
import { Grid, OrbitControls } from '@react-three/drei';
import * as THREE from 'three';
import { CATALOG } from '../../catalog';
import { computeConnections } from '../../model/connections';
import { DIRS, addVec, posKey } from '../../model/directions';
import { placementFacing, type BlockMap } from '../../model/editing';
import type { Facing, Vec3 } from '../../model/types';
import { ConduitMesh } from './ConduitMesh';
import { MachineMesh } from './MachineMesh';
import { isClick, sideFromNormal } from './geometry';

export interface Hover { pos: Vec3; side: number; ground: boolean }

export interface SceneProps {
  blocks: BlockMap;
  layer: number | null;
  selectedKey: string;
  hover: Hover | null;
  onHover(h: Hover | null): void;
  onPlace(pos: Vec3, facing: Facing): void;
  onRemove(pos: Vec3): void;
  onToggleSide(pos: Vec3, side: number): void;
}

function hoverFromEvent(e: ThreeEvent<PointerEvent | MouseEvent>): Hover | null {
  const pos = e.object.userData.pos as Vec3 | undefined;
  const normal = e.face?.normal;
  if (!pos || !normal) return null;
  const side = sideFromNormal([normal.x, normal.y, normal.z]);
  return side < 0 ? null : { pos, side, ground: false };
}

function groundHover(e: ThreeEvent<PointerEvent | MouseEvent>): Hover {
  return { pos: [Math.floor(e.point.x), -1, Math.floor(e.point.z)], side: 1, ground: true };
}

function Blocks(props: SceneProps) {
  const { camera } = useThree();
  const { blocks, layer } = props;
  const visible = useMemo(() => [...blocks.values()].filter((b) => layer === null || b.pos[1] <= layer), [blocks, layer]);
  const masks = useMemo(() => computeConnections([...blocks.values()]), [blocks]);

  const place = (h: Hover) => {
    const forward = camera.getWorldDirection(new THREE.Vector3());
    props.onPlace(addVec(h.pos, DIRS[h.side]), placementFacing(h.side, [forward.x, forward.y, forward.z]));
  };

  const onBlockClick = (e: ThreeEvent<MouseEvent>) => {
    e.stopPropagation();
    if (!isClick(e.delta)) return;
    const h = hoverFromEvent(e);
    if (!h) return;
    const entry = CATALOG.get(blocks.get(posKey(h.pos))?.key ?? '');
    if (e.shiftKey && entry && entry.kind !== 'machine') props.onToggleSide(h.pos, h.side);
    else place(h);
  };
  const onBlockContext = (e: ThreeEvent<MouseEvent>) => {
    e.stopPropagation();
    e.nativeEvent.preventDefault();
    if (!isClick(e.delta)) return;
    const pos = e.object.userData.pos as Vec3 | undefined;
    if (pos) props.onRemove(pos);
  };

  return (
    <>
      <group
        onPointerMove={(e) => { e.stopPropagation(); props.onHover(hoverFromEvent(e)); }}
        onPointerOut={() => props.onHover(null)}
        onClick={onBlockClick}
        onContextMenu={onBlockContext}
      >
        {visible.map((b) => {
          const entry = CATALOG.get(b.key);
          if (!entry) {
            return (
              <mesh key={posKey(b.pos)} position={[b.pos[0] + 0.5, b.pos[1] + 0.5, b.pos[2] + 0.5]} userData={{ pos: b.pos }}>
                <boxGeometry args={[1, 1, 1]} />
                <meshStandardMaterial color="#ff00ff" />
              </mesh>
            );
          }
          return entry.kind === 'machine'
            ? <MachineMesh key={posKey(b.pos)} block={b} entry={entry} />
            : <ConduitMesh key={posKey(b.pos)} block={b} entry={entry} mask={masks.get(posKey(b.pos)) ?? 0} />;
        })}
      </group>
      <mesh
        rotation={[-Math.PI / 2, 0, 0]}
        onPointerMove={(e) => { e.stopPropagation(); props.onHover(groundHover(e)); }}
        onPointerOut={() => props.onHover(null)}
        onClick={(e) => { e.stopPropagation(); if (isClick(e.delta)) place(groundHover(e)); }}
      >
        <planeGeometry args={[400, 400]} />
        <meshBasicMaterial visible={false} />
      </mesh>
      <Ghost hover={props.hover} blocks={blocks} />
    </>
  );
}

/** Wireframe on the hovered block plus a translucent preview where the next block would go. */
function Ghost({ hover, blocks }: { hover: Hover | null; blocks: BlockMap }) {
  if (!hover) return null;
  const target = addVec(hover.pos, DIRS[hover.side]);
  const free = !blocks.has(posKey(target));
  return (
    <>
      {!hover.ground && (
        <mesh position={[hover.pos[0] + 0.5, hover.pos[1] + 0.5, hover.pos[2] + 0.5]} raycast={() => undefined}>
          <boxGeometry args={[1.02, 1.02, 1.02]} />
          <meshBasicMaterial color="#ffd54f" wireframe />
        </mesh>
      )}
      {free && (
        <mesh position={[target[0] + 0.5, target[1] + 0.5, target[2] + 0.5]} raycast={() => undefined}>
          <boxGeometry args={[1, 1, 1]} />
          <meshBasicMaterial color="#4fc3f7" transparent opacity={0.25} depthWrite={false} />
        </mesh>
      )}
    </>
  );
}

export function Scene(props: SceneProps) {
  return (
    <Canvas camera={{ position: [10, 9, 12], fov: 50 }} onContextMenu={(e) => e.preventDefault()}>
      <color attach="background" args={['#1e2027']} />
      <ambientLight intensity={0.9} />
      <directionalLight position={[8, 12, 6]} intensity={1.1} />
      <Grid args={[200, 200]} cellSize={1} sectionSize={8} cellColor="#3a3d47" sectionColor="#555a66" infiniteGrid fadeDistance={80} />
      <Blocks {...props} />
      <OrbitControls
        makeDefault
        mouseButtons={{ LEFT: THREE.MOUSE.ROTATE, MIDDLE: THREE.MOUSE.PAN, RIGHT: THREE.MOUSE.PAN }}
      />
    </Canvas>
  );
}
```

```tsx
import { useCallback, useEffect, useMemo, useReducer, useRef, useState } from 'react';
import { buildSchematic } from './export/schematic';
import { computeConnections } from './model/connections';
import { posKey } from './model/directions';
import { billOfMaterials } from './model/editing';
import { parseBuild, serializeBuild } from './model/buildJson';
import { countsForBerryRate, makeOilBerryLine } from './presets/oilBerryLine';
import { downloadBytes, downloadText, safeFileName } from './editor/download';
import { loadFromStorage, safeStorage, saveToStorage } from './editor/persistence';
import { initialState, reducer, toBuild } from './editor/state';
import { Scene, type Hover } from './editor/scene/Scene';
import { InfoPanel } from './editor/ui/InfoPanel';
import { MaterialsPanel } from './editor/ui/MaterialsPanel';
import { Palette } from './editor/ui/Palette';
import { Toasts, type Toast } from './editor/ui/Toasts';
import { Toolbar } from './editor/ui/Toolbar';

const restored = loadFromStorage(safeStorage());

export function App() {
  const [state, dispatch] = useReducer(reducer, restored.build ?? undefined, initialState);
  const [hover, setHover] = useState<Hover | null>(null);
  const [toasts, setToasts] = useState<Toast[]>([]);
  const nextId = useRef(1);

  const toast = useCallback((text: string, kind: Toast['kind'] = 'info') => {
    const id = nextId.current++;
    setToasts((t) => [...t, { id, text, kind }]);
    setTimeout(() => setToasts((t) => t.filter((x) => x.id !== id)), 6000);
  }, []);

  useEffect(() => {
    if (restored.error) toast(restored.error, 'error');
  }, [toast]);

  // Debounced autosave.
  useEffect(() => {
    const h = setTimeout(() => saveToStorage(safeStorage(), toBuild(state)), 400);
    return () => clearTimeout(h);
  }, [state.blocks, state.name]); // eslint-disable-line react-hooks/exhaustive-deps

  // R / Shift+R rotate the hovered machine.
  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if ((e.target as HTMLElement).tagName === 'INPUT') return;
      if (e.key.toLowerCase() === 'r' && hover && !hover.ground) {
        dispatch({ type: 'rotate', pos: hover.pos, includeVertical: e.shiftKey });
      }
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [hover]);

  const bom = useMemo(() => billOfMaterials(state.blocks.values()), [state.blocks]);
  const maxY = useMemo(() => Math.max(0, ...[...state.blocks.values()].map((b) => b.pos[1])), [state.blocks]);
  const hovered = hover && !hover.ground ? state.blocks.get(posKey(hover.pos)) : undefined;
  const hoveredMask = useMemo(
    () => (hovered ? computeConnections([...state.blocks.values()]).get(posKey(hovered.pos)) : undefined),
    [hovered, state.blocks],
  );

  const onExport = () => {
    try {
      const { bytes, warnings } = buildSchematic(toBuild(state));
      downloadBytes(bytes, `${safeFileName(state.name)}.schematic`);
      toast(`Exported ${safeFileName(state.name)}.schematic — copy it into .minecraft/schematics`);
      for (const w of warnings) toast(w, 'error');
    } catch (err) {
      toast((err as Error).message, 'error');
    }
  };

  const onExample = () => {
    const answer = window.prompt('Berries per second to size the line for?', '5.6');
    if (answer === null) return;
    const rate = Number(answer);
    try {
      if (!(rate > 0)) throw new Error('Enter a positive number of berries per second');
      const build = makeOilBerryLine(countsForBerryRate(rate));
      dispatch({ type: 'load', build });
      toast(`Loaded ${build.name}`);
    } catch (err) {
      toast((err as Error).message, 'error');
    }
  };

  const onLoadFile = (text: string) => {
    try {
      const { build, warnings } = parseBuild(text);
      dispatch({ type: 'load', build });
      toast(`Loaded ${build.name}`);
      for (const w of warnings) toast(w, 'error');
    } catch (err) {
      toast((err as Error).message, 'error');
    }
  };

  return (
    <div className="app">
      <Toolbar
        name={state.name}
        canExport={state.blocks.size > 0}
        maxY={maxY}
        layer={state.layer}
        onName={(name) => dispatch({ type: 'setName', name })}
        onNew={() => window.confirm('Clear the current build?') && dispatch({ type: 'new' })}
        onExample={onExample}
        onSave={() => downloadText(serializeBuild(toBuild(state)), `${safeFileName(state.name)}.json`)}
        onLoadFile={onLoadFile}
        onExport={onExport}
        onLayer={(layer) => dispatch({ type: 'setLayer', layer })}
      />
      <Palette selectedKey={state.selectedKey} onSelect={(key) => dispatch({ type: 'select', key })} />
      <main className="viewport">
        <Scene
          blocks={state.blocks}
          layer={state.layer}
          selectedKey={state.selectedKey}
          hover={hover}
          onHover={setHover}
          onPlace={(pos, facing) => dispatch({ type: 'place', pos, facing })}
          onRemove={(pos) => dispatch({ type: 'remove', pos })}
          onToggleSide={(pos, side) => dispatch({ type: 'toggleSide', pos, side })}
        />
      </main>
      <div className="side">
        <InfoPanel block={hovered} mask={hoveredMask} />
        <MaterialsPanel lines={bom} />
      </div>
      <Toasts toasts={toasts} onDismiss={(id) => setToasts((t) => t.filter((x) => x.id !== id))} />
    </div>
  );
}
```

```ts
import { expect, test } from '@playwright/test';
import { readFileSync } from 'node:fs';

test.beforeEach(async ({ page }) => {
  await page.goto('/');
  await page.evaluate(() => localStorage.clear());
  await page.reload();
});

test('place a machine, see it in the materials list, export a gzipped schematic', async ({ page }) => {
  await page.getByTestId('palette-distillery_lv').click();
  const canvas = page.locator('canvas');
  const box = (await canvas.boundingBox())!;
  await page.mouse.click(box.x + box.width / 2, box.y + box.height * 0.6);
  await expect(page.getByTestId('materials')).toContainText('1× LV Distillery');

  const download = page.waitForEvent('download');
  await page.getByTestId('export').click();
  const file = await (await download).path();
  const bytes = readFileSync(file!);
  expect(bytes[0]).toBe(0x1f);
  expect(bytes[1]).toBe(0x8b);
});

test('right-click removes, and the build survives a reload', async ({ page }) => {
  const canvas = page.locator('canvas');
  const box = (await canvas.boundingBox())!;
  const at = { x: box.x + box.width / 2, y: box.y + box.height * 0.6 };
  await page.mouse.click(at.x, at.y);
  await expect(page.getByTestId('materials')).toContainText('1× LV Distillery');
  await page.mouse.click(at.x, at.y - 8, { button: 'right' }); // a few px up: lands on the placed block
  await expect(page.getByTestId('materials')).toHaveCount(0);
  await page.mouse.click(at.x, at.y);
  await expect(page.getByTestId('materials')).toContainText('1× LV Distillery');
  await page.waitForTimeout(600); // autosave debounce
  await page.reload();
  await expect(page.getByTestId('materials')).toContainText('1× LV Distillery');
});

test('loads the oil berry example', async ({ page }) => {
  page.once('dialog', (d) => d.accept('5.6'));
  await page.getByTestId('example').click();
  await expect(page.getByTestId('materials')).toContainText('6× LV Fluid Extractor');
  await expect(page.getByTestId('materials')).toContainText('17× LV Distillery');
  await expect(page.getByTestId('export')).toBeEnabled();
});

test('a dragged pointer does not place anything', async ({ page }) => {
  const canvas = page.locator('canvas');
  const box = (await canvas.boundingBox())!;
  const x = box.x + box.width / 2, y = box.y + box.height * 0.6;
  await page.mouse.move(x, y);
  await page.mouse.down();
  await page.mouse.move(x + 80, y + 20, { steps: 8 });
  await page.mouse.up();
  await expect(page.getByTestId('materials')).toHaveCount(0);
  await expect(page.getByTestId('export')).toBeDisabled();
});
```
